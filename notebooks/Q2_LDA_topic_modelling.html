Denoising steps: strip HTML tags and URLs, lower-case, remove digits and punctuation, collapse whitespace, remove stop words and very short tokens, lemmatise.

clean_text <- function(x) {
  x |>
    str_replace_all("<[^>]+>", " ") |>           # HTML
    str_replace_all("http\\S+|www\\.\\S+", " ") |> # URLs
    str_replace_all("\\S+@\\S+", " ") |>         # e-mails
    str_to_lower() |>
    str_replace_all("[^a-z\\s]", " ") |>         # punctuation, digits, symbols
    str_squish()
}

news_df <- news_df |> mutate(clean = clean_text(content))

extra_stop <- c("said","say","says","will","also","would","could","one","two","new","year","years","mr","mrs")

tokens <- news_df |>
  select(file_name, clean) |>
  unnest_tokens(word, clean) |>
  filter(!word %in% stop_words$word,
         !word %in% extra_stop,
         nchar(word) > 2) |>
  mutate(word = lemmatize_words(word))

# lemmatising can reintroduce stop words, so filter again
tokens <- tokens |> filter(!word %in% stop_words$word)

# remove very rare words (< 3 occurrences overall)
tokens <- tokens |> group_by(word) |> filter(n() >= 3) |> ungroup()

# preprocessed data frame (file_name, content)
news_clean <- tokens |>
  group_by(file_name) |>
  summarise(content = paste(word, collapse = " "), .groups = "drop")

2.2 LDA

Document-term matrix and model:

dtm <- tokens |>
  count(file_name, word) |>
  cast_dtm(file_name, word, n)

lda_model <- topicmodels::LDA(dtm, k = N_TOPICS, method = "Gibbs",
                 control = list(seed = 1234, burnin = 500, iter = 2000))
class(lda_model)
## [1] "LDA_Gibbs"
## attr(,"package")
## [1] "topicmodels"
# posterior distributions (topic-word = beta, document-topic = gamma)
post <- topicmodels::posterior(lda_model)

Top k words per topic (beta = probability of the word given the topic, i.e. the word distribution):

topic_words <- as_tibble(t(post$terms), rownames = "term") |>
  pivot_longer(-term, names_to = "topic", values_to = "beta") |>
  mutate(topic = as.integer(topic))

top_terms <- topic_words |>
  group_by(topic) |>
  slice_max(beta, n = K_WORDS, with_ties = FALSE) |>
  ungroup() |>
  arrange(topic, desc(beta))

Word distribution reporting: how concentrated each topic is, and how topics spread over documents.

# share of each topic's probability mass held by its top k words
topic_words |>
  group_by(topic) |>
  arrange(desc(beta), .by_group = TRUE) |>
  summarise(top_k_mass = sum(head(beta, K_WORDS)), vocab = n())

2.3 Visualisation

Method 1 - faceted bar chart of top words per topic

top_terms |>
  mutate(term = reorder_within(term, beta, topic)) |>
  ggplot(aes(beta, term, fill = factor(topic))) +
  geom_col(show.legend = FALSE) +
  facet_wrap(~ paste("Topic", topic), scales = "free_y") +
  scale_y_reordered() +
  labs(title = "Top words per topic (LDA)",
       x = "Word probability within topic (beta)", y = "Word")

Method 2 - word clouds per topic

par(mfrow = c(ceiling(N_TOPICS / 2), 2), mar = c(0, 0, 2, 0))
for (t in seq_len(N_TOPICS)) {
  d <- filter(top_terms, topic == t)
  wordcloud(d$term, d$beta, scale = c(3, 0.6), min.freq = 0,
            colors = brewer.pal(8, "Dark2"), random.order = FALSE)
  title(paste("Topic", t))
}
